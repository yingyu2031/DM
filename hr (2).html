💻 步驟三：Google Apps Script 一體化串接程式碼（支援分科計分）

自動解析 Google 表單的 [人事]/[考勤]/[薪資] 題目，並將各科獨立分數寫回 Notion 資料庫

/**
 * HR 系統輔導測驗平台 - Google Apps Script 一體化後端 (支援人事/考勤/薪資分科計分)
 * 功能 1 (doGet)：讀取 Google 表單題目，自動識別 [人事]、[考勤]、[薪資] 分類並回傳 JSON
 * 功能 2 (doPost)：接收客戶交卷結果，將各科獨立分數寫入 Notion 資料庫（未考科目自動留空）
 */

// ====== 請填入以下三個設定值 ======
var GOOGLE_FORM_ID = "請填入您的Google表單ID"; // 表單網址 /d/ 與 /edit 之間的那串英數字
var NOTION_TOKEN   = "secret_請填入您的Notion整合金鑰";
var NOTION_DB_ID   = "請填入您的Notion資料庫ID";

// 1. 客戶開啟網頁時，自動抓取 Google 表單題目與科別分類
function doGet(e) {
  try {
    var form = FormApp.openById(GOOGLE_FORM_ID);
    var items = form.getItems();
    var optionLetters = ["A", "B", "C", "D", "E", "F"];
    var questions = [];

    items.forEach(function(item) {
      var type = item.getType();
      if (type !== FormApp.ItemType.MULTIPLE_CHOICE && type !== FormApp.ItemType.CHECKBOX) {
        return;
      }

      var rawTitle = item.getTitle() || "";
      var category = "薪資"; // 預設歸類
      var subTopic = "實務操作";
      var cleanTitle = rawTitle;

      // 解析 [科別][小單元] 題目內容，例如：[人事][到離職異動] 新進員工...
      var tagMatch = rawTitle.match(/^\[(人事|考勤|薪資)\](?:\[(.+?)\])?\s*(.*)$/);
      if (tagMatch) {
        category = tagMatch[1];
        if (tagMatch[2]) subTopic = tagMatch[2];
        cleanTitle = tagMatch[3];
      }

      var choices = [];
      var feedback = null;

      if (type === FormApp.ItemType.MULTIPLE_CHOICE) {
        var mcItem = item.asMultipleChoiceItem();
        choices = mcItem.getChoices();
        feedback = mcItem.getFeedbackForCorrect();
      } else if (type === FormApp.ItemType.CHECKBOX) {
        var cbItem = item.asCheckboxItem();
        choices = cbItem.getChoices();
        feedback = cbItem.getFeedbackForCorrect();
      }

      var optionsMap = {};
      var correctAnswers = [];

      choices.forEach(function(choice, idx) {
        if (idx < optionLetters.length) {
          var letter = optionLetters[idx];
          optionsMap[letter] = choice.getValue();
          if (choice.isCorrectAnswer()) {
            correctAnswers.push(letter);
          }
        }
      });

      if (correctAnswers.length === 0) correctAnswers = ["A"];

      questions.push({
        id: questions.length + 1,
        category: category,
        subTopic: subTopic,
        type: (type === FormApp.ItemType.CHECKBOX || correctAnswers.length > 1) ? "multiple" : "single",
        question: cleanTitle,
        options: optionsMap,
        answer: correctAnswers,
        explanation: feedback ? feedback.getText() : "請參考 HR 系統標準作業手冊之相關設定說明。"
      });
    });

    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      questions: questions
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

// 2. 客戶交卷送出後，將各科獨立成績寫回 Notion 資料庫
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var scores = data.scoresByCategory || {};
    var selectedCats = (data.selectedCategories || []).map(function(cat) {
      return { name: cat };
    });

    var url = "https://api.notion.com/v1/pages";

    // 未選考的科目為 null，寫入 Notion Number 欄位時設為 null 即可保持空白不填 0
    var payload = {
      parent: { database_id: NOTION_DB_ID },
      properties: {
        "公司抬頭": { title: [{ text: { content: data.company || "未填寫" } }] },
        "應試人員": { rich_text: [{ text: { content: data.name || "" } }] },
        "測驗次數": { number: Number(data.attempt || 1) },
        "選考科目": { multi_select: selectedCats },
        "人事分數": { number: (scores["人事"] !== null && scores["人事"] !== undefined) ? Number(scores["人事"]) : null },
        "考勤分數": { number: (scores["考勤"] !== null && scores["考勤"] !== undefined) ? Number(scores["考勤"]) : null },
        "薪資分數": { number: (scores["薪資"] !== null && scores["薪資"] !== undefined) ? Number(scores["薪資"]) : null },
        "總平均分": { number: Number(data.overallScore || 0) },
        "作答明細": { rich_text: [{ text: { content: (data.details || "").substring(0, 1900) } }] },
        "測驗時間": { date: { start: new Date().toISOString() } }
      }
    };

    var res = UrlFetchApp.fetch(url, {
      method: "post",
      contentType: "application/json",
      headers: {
        "Authorization": "Bearer " + NOTION_TOKEN,
        "Notion-Version": "2022-06-28"
      },
      payload: JSON.stringify(payload),
      muteHttpExceptions: true
    });

    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      notionResponse: res.getContentText()
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
